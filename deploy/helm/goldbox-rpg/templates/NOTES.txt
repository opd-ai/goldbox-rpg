1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "goldbox-rpg.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "goldbox-rpg.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "goldbox-rpg.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "goldbox-rpg.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Check the health status:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  curl http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}/health
{{- end }}
{{- else }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "goldbox-rpg.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} exec $POD_NAME -- wget -qO- http://localhost:8080/health
{{- end }}

3. View the logs:
  kubectl logs --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "goldbox-rpg.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -f

4. Monitor Prometheus metrics:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  curl http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}/metrics
{{- end }}
{{- else }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "goldbox-rpg.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} exec $POD_NAME -- wget -qO- http://localhost:8080/metrics
{{- end }}

{{- if .Values.persistence.enabled }}

5. Persistence is enabled. Your data will be stored in:
  PVC: {{ include "goldbox-rpg.fullname" . }}-data
  Mount Path: {{ .Values.persistence.mountPath }}
  
  To view the PVC status:
  kubectl get pvc --namespace {{ .Release.Namespace }} {{ include "goldbox-rpg.fullname" . }}-data
{{- else }}

5. WARNING: Persistence is disabled. Data will be lost on pod restarts!
  To enable persistence, set persistence.enabled=true in your values.yaml
{{- end }}

6. Autoscaling is enabled:

{{- if not .Values.ingress.enabled }}

7. IMPORTANT: Ingress is disabled. To expose the application externally:
  - Enable ingress in values.yaml, OR
  - Change service.type to LoadBalancer, OR
  - Use kubectl port-forward (shown above)
{{- end }}
